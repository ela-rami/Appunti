Step 6: Client (Main)

Il `Main` funge da client finale. Deve dimostrare l'uso dell'API in modo pulito, senza mai accedere ai componenti interni del sistema.

Main.java

public class Main {
    public static void main(String[] args) {
        System.out.println("--- Sistema di Noleggio Barche ---");
        API api = new API();
        NoleggioFactory factory = new NoleggioFactory();

        // --- Scenario 1: Un cliente cerca e prenota una barca ---
        System.out.println("\n[SCENARIO 1] Ricerca e Prenotazione");
        
        Cliente cliente = factory.creaCliente("C001", "Mario Rossi");
        PeriodoNoleggio periodo = factory.creaPeriodo(
            java.time.LocalDateTime.now(), 
            java.time.LocalDateTime.now().plusDays(7)
        );

        CriterioRicercaBarca criterio = b -> b.getDimensioneMetri() >= 12;
        java.util.List barcheTrovate = api.cercaDisponibilita(criterio, periodo);
        System.out.println("Trovate " + barcheTrovate.size() + " barche disponibili.");

        if (!barcheTrovate.isEmpty()) {
            Barca barcaScelta = barcheTrovate.get(0);
            
            try {
                java.util.List extra = java.util.List.of(TipoExtra.SKIPPER);
                Noleggio noleggio = api.prenota(barcaScelta, cliente, periodo, extra);
                System.out.println("Prenotazione confermata! ID: " + noleggio.getId());
                System.out.printf("Costo Totale Calcolato: %.2f €\n", noleggio.getCostoFinale());
            } catch (BarcaNonDisponibileException e) {
                System.err.println("Prenotazione fallita: " + e.getMessage());
            }
        }
    }
}
